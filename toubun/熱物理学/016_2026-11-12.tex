\toi{1}{◎}{}
\[ \left(\pdv{X}{T}\right)_{V}=\frac{2T}{V}, \qquad \left(\pdv{X}{V}\right)_{T}=-\frac{T^{2}}{V^{2}} \]
\begin{align*}
\dd X&=\left(\pdv{X}{T}\right)_{V}\dd T+\left(\pdv{X}{V}\right)_{T}\dd V\\
&=\frac{2T}{V}\dd T-\frac{T^{2}}{V^{2}}\dd V
\end{align*}
\toi{2}{◎}{}
\[ V=\frac{nRT}{p} \]
\[ X(T,p)=\frac{pT^{2}}{nRT}=\frac{pT}{nR} \]
\[ \left(\pdv{X}{T}\right)_{p}=\frac{p}{nR}=\frac{T}{V} \]
\[ \left(\pdv{X}{p}\right)_{T}=\frac{T}{nR} \]
\toi{3}{◎}{}
\[ \left(\pdv{X}{T}\right)_{p}=\frac{T}{V} : \left(\pdv{X}{T}\right)_{V}=\frac{2T}{V} \]
\[ \therefore\ 1:2 \]
