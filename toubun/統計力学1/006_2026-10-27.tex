\toi{1}{◎}{}
\[ U=R^{2}\kakimachigai{E}{E_{0}},\quad\therefore\ R=\sqrt{\frac{U}{\kakimachigai{E}{E_{0}}}}=\frac{\sqrt{2mL^{2}U}}{\pi\hbar} \]
\begin{align*}
\delta R&=\dv{R}{U}\,\delta U=\frac{2mL^{2}\,\delta U}{2\pi\hbar\sqrt{2mL^{2}U}}\\
&=\frac{\sqrt{2mL^{2}U}}{\pi\hbar}\,\frac{\delta U}{2U}\\
&=\frac{R}{2U}\,\delta U
\end{align*}
\toi{2}{◎}{}
\begin{align*}
w&=\frac{C(2N)}{2^{2N}N!}\left((R+\delta R)^{2N}-R^{2N}\right)\\
&=\frac{\pi^{N}}{2^{2N}N!\,\Gamma(N+1)}\biggl(R^{2N}\left(1+\frac{\delta U}{2U}\right)^{2N}\\
&\qquad-R^{2N}\biggr)\\
&=\frac{\pi^{N}}{2^{2N}(N!)^{2}}R^{2N}\biggl(\Bigl(1+N\frac{\delta U}{U}\\
&\qquad+O\Bigl(\Bigl(\frac{\delta U}{2U}\Bigr)^{2}\Bigr)\Bigr)-1\biggr)\\
&=\frac{\pi^{N}(2mL^{2}U)^{N}}{2^{2N}(N!)^{2}\pi^{2N}\hbar^{2N}}\,N\frac{\delta U}{U}\\
&=\frac{N(2\pi mU)^{N}A^{N}}{(N!)^{2}h^{2N}}\,\frac{\delta U}{U}
\end{align*}
\toi{3}{◎}{}
\[ S=\kB\log\frac{N(2\pi mU)^{N}A^{N}}{(N!)^{2}h^{2N}}\,\frac{\delta U}{U} \]
（$\log N$ と $\log(\delta U/U)$ は斜線で消して落とす）
\begin{align*}
&=\kB\bigl[N\log(2\pi mU)+N\log A\\
&\qquad-2\log N!-2N\log h\bigr]\\
&=\kB N\bigl[\log(2\pi mU)+\log A-2\log N\\
&\qquad+2-2\log h\bigr]\\
&=\kB N\left[2\log\frac{1}{Nh}+\log(2\pi mUA)+2\right]
\end{align*}
\toi{4}{◎}{}
\[ \frac{1}{T}=\left(\pdv{S}{U}\right)_{A}=\frac{\kB N}{U}\qquad\therefore\ U=\kB NT \]
\[ C_{A}=\left(\pdv{U}{T}\right)_{A}=\kB N \]
\[ \frac{\Pi}{T}=\left(\pdv{S}{A}\right)_{U}=\frac{\kB N}{A}\qquad\Pi A=\kB NT \]
